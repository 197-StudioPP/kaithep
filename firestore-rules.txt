rules_version = '2';
// วางทั้งหมดนี้ใน Firebase Console → Firestore Database → Rules แล้วกด Publish
service cloud.firestore {
  match /databases/{database}/documents {

    // เซฟของแต่ละคน: อ่าน/เขียนได้เฉพาะเจ้าของ
    match /saves/{uid} {
      allow read, write: if request.auth != null && request.auth.uid == uid;
    }

    // ตัวแทน AI: ทุกคนที่ล็อกอินอ่านได้ แต่เขียนได้เฉพาะเจ้าของ
    match /ghosts/{uid} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && request.auth.uid == uid;
    }

    // บันทึกการชิงไก่: สร้างได้ครั้งเดียวต่อไก่ 1 ตัว (ห้ามแก้ทับ = คนแรกที่ชนะได้ไป)
    match /steals/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
                    && request.resource.data.from == request.auth.uid
                    && request.resource.data.to != request.auth.uid
                    && id == request.resource.data.to + '_' + request.resource.data.u;
      allow delete: if request.auth != null && resource.data.to == request.auth.uid;
    }

    // ใบประกาศ: ใครสู้กับใคร — ทุกคนที่ล็อกอินอ่านได้ เพิ่มได้เฉพาะในชื่อตัวเอง ห้ามแก้/ลบ
    match /feed/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null && request.resource.data.uid == request.auth.uid;
    }

    // ปล้นไข่ (แพตช์ 1.1.3): คนปล้นสร้างได้ในชื่อตัวเอง • เจ้าของฟาร์มอ่านแล้วลบทิ้ง
    match /raids/{id} {
      allow create: if request.auth != null
                    && request.resource.data.from == request.auth.uid
                    && request.resource.data.to != request.auth.uid
                    && request.resource.data.n <= 25;
      allow read, delete: if request.auth != null && resource.data.to == request.auth.uid;
    }
  }
}
